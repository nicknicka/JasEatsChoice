## 1️⃣ 获取个性化推荐（核心功能）

**位置**: `useRecommendations.js:240-267`

### API调用
```javascript
// 📍 GET /v1/recommend/recommend/{userId}
const response = await axios.get(`${API_CONFIG.baseURL}/v1/recommend/recommend/${userId}`)
```

### 代码实现
```javascript
const fetchRecommendationsFromBackend = async (rejectedDishIds) => {
  try {
    // 获取用户ID
    const userId = parseInt(String(authStore.userId || 1) || '1', 10)

    // 🔴 向后端发送请求：获取个性化推荐
    const response = await axios.get(`${API_CONFIG.baseURL}/v1/recommend/recommend/${userId}`)

    const data = response.data.data
    if (data && data.dishes) {
      // 过滤掉被拒绝的菜品
      const personalizedRecs = data.dishes
        .filter(dish => {
          // 检查是否被拒绝过
          if (!rejectedDishIds || rejectedDishIds.length === 0) return true
          return !isRejected(rejectedDishIds, dish.dishId || dish.id)
        })
        .map((dish, index) => ({
          ...dish,
          id: Date.now() + index + 2000,
          dishId: dish.dishId || dish.id || String(Date.now() + index + 2000),
          recommendSource: RECOMMENDATION_TYPES.PERSONALIZED,
          reason: dish.reason || '基于您的饮食偏好推荐'
        }))

      // 为标签分配随机类型
      assignRandomTagTypes(personalizedRecs)
      return personalizedRecs
    }
    return []
  } catch (error) {
    console.error('获取推荐数据失败:', error)
    return null
  }
}
```

### 返回数据结构
```json

  "code": 200,
  "message": "success",
  "data": {

      {
        "dishId": "1001",
```

```javascript
// 📍 GET /v1/recommend/reject/list?userId={userId}

```

### 代码实现
```javascript
const loadRejectionHistory = async () => {
  try {
    const userId = String(authStore.userId || 1)

    // 🔴 向后端发送请求：获取已拒绝的菜品列表
    const response = await axios.get(`${API_CONFIG.baseURL}${API_CONFIG.recommendReject.list}`, {
      params: { userId }
    })

    // 返回已拒绝的菜品ID列表
    return response.data.data || []
  } catch (error) {
    console.error('加载拒绝历史失败:', error)
    // 🛡️ 降级策略：如果后端失败，从 localStorage 读取
    const saved = localStorage.getItem('rejectionHistory')
    return saved ? JSON.parse(saved) : []
  }
}
```

### 返回数据结构
```json
{
  "code": 200,
  "message": "success",
  "data": ["1001", "1002", "1003"]
}
```

### 调用时机
```javascript
// 在 loadAllRecommendations() 开始时调用
const loadAllRecommendations = async () => {
```

## 3️⃣ 记录拒绝推荐（写入操作）

**位置**: `useRecommendations.js:52-80`

### API调用
```javascript
// 📍 POST /v1/recommend/reject?userId={userId}&dishId={dishId}
await axios.post(`${API_CONFIG.baseURL}${API_CONFIG.recommendReject.add}`, null, {
  params: { userId, dishId }
})
```

### 代码实现
```javascript
const saveRejectionHistory = async (userId, dishId) => {
  try {
    // 🔴 向后端发送请求：记录拒绝
    await axios.post(`${API_CONFIG.baseURL}${API_CONFIG.recommendReject.add}`, null, {
      params: { userId, dishId }
    })

    // 成功后清除本地缓存
    localStorage.removeItem('rejectionHistory')
  } catch (error) {
    console.error('保存拒绝历史失败:', error)

    // 🛡️ 降级策略：保存到 localStorage
    let history = []
    const saved = localStorage.getItem('rejectionHistory')
    if (saved) {
      history = JSON.parse(saved)
    }

    const existingIndex = history.findIndex((entry) => entry.dishId === dishId)
    if (existingIndex > -1) {
      history[existingIndex].count += 1
    } else {
      history.push({
        dishId,
        count: 1,
        rejectedAt: new Date().toISOString()
      })
    }

    localStorage.setItem('rejectionHistory', JSON.stringify(history))
  }
}
```

### 调用时机
```javascript
// 用户点击"不感兴趣"按钮时

  const userId = String(authStore.userId || 1)

}
```

### 代码实现
```javascript
const updateRecommendationsByWeatherAndTime = async (rejectedDishIds) => {
  try {
    // 🔴 向后端发送请求：获取天气信息
    const response = await axios.get(API_CONFIG.baseURL + API_CONFIG.weather.current, {
      params: { city: '北京' }
    })

    const weatherData = response.data.data
    const { temperature, humidity } = weatherData

    // 根据天气生成推荐标签
    const timeType = getCurrentTimePeriod()
    let weatherTags = []
    if (temperature > 30) weatherTags.push('冰饮', '凉菜', '轻食')
    else if (temperature < 10) weatherTags.push('热饮', '热菜', '火锅')
    if (humidity > 80) weatherTags.push('祛湿粥品', '清淡饮食')

    return generateWeatherTimeRecommendations(timeType, weatherTags, rejectedDishIds)
  } catch (error) {
    console.error('天气推荐失败:', error)
    return []
  }
}
```

```javascript

  fetchRecommendationsFromBackend(rejectedDishIds),
```

## 🎯 总结：哪些数据从后端获取？

| 数据类型 | API端点 | 请求方法 | 用途 |
|---------|---------|----------|------|
| **个性化推荐** | `/v1/recommend/recommend/{userId}` | GET | 获取基于用户画像的智能推荐 |
| **拒绝历史** | `/v1/recommend/reject/list?userId={userId}` | GET | 获取用户拒绝过的菜品列表 |
| **天气数据** | `/v1/weather?city={city}` | GET | 获取天气用于生成推荐 |
| **记录拒绝** | `/v1/recommend/reject?userId={userId}&dishId={dishId}` | POST | 保存用户拒绝记录 |

---

## 🔍 关键点

### 1. 拒绝历史的作用
- **从后端加载** → 获取用户之前拒绝的菜品列表
- **用于过滤推荐** → 在生成推荐时，自动过滤掉这些菜品
- **双重过滤** → 前端过滤 + 后端过滤，确保不推荐被拒绝的菜品

### 2. 降级策略
- 所有 `GET` 请求失败时，都会降级到 `localStorage`
- 所有 `POST` 请求失败时，也会保存到 `localStorage`
- 确保即使后端故障，功能仍可用

### 3. 数据流
```
后端数据库 ←→ 前端页面 ←→ 用户操作
    ↓                              ↓
 提供数据                      收集反馈
```

## 💡 最关键的部分

**`fetchRecommendationsFromBackend()` 函数** - 这是从后端获取推荐数据的核心函数：

```javascript
const response = await axios.get(`${API_CONFIG.baseURL}/v1/recommend/recommend/${userId}`)
```

这个API调用会触发后端的推荐引擎：
1. 获取用户画像
2. 执行四种召回策略（用户画像、协同过滤、热门菜品、上下文）
3. 过滤掉被拒绝>=2次的菜品
4. 排序和多样性处理
5. 生成推荐理由（使用AI）
6. 返回最终推荐列表

这就是整个推荐系统的核心数据获取逻辑！🎯
