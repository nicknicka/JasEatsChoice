### 1.3 数据加载优化

#### 当前问题
- 多个串行请求导致加载时间长
- 缺少请求合并和批处理

#### 优化方案

**并行加载优化**
```javascript
// 当前实现（串行）
onMounted(async () => {
  await loadBanners()
  await loadMerchants()
  await loadHotTopic()
  await loadRecommendations()
})

// 优化后（并行）
onMounted(async () => {
  await Promise.all([
    loadBanners(),
    loadMerchants(),
    loadHotTopic()
  ])

  // 推荐系统依赖用户信息，稍后加载
  await loadRecommendations()
})
```

**实现请求合并**
```javascript
/**
 * 首页数据聚合接口
 */
const loadHomePageData = async () => {
  const res = await uni.request({
    url: '/api/v1/home/data',
    method: 'GET',
    data: {
      userId: userStore.userInfo.userId,
      latitude: locationStore.latitude,
      longitude: locationStore.longitude
    }
  })

  // 一次性获取所有数据
  const { banners, merchants, hotTopic, categories } = res.data

  return { banners, merchants, hotTopic, categories }
}
```

**预加载策略**
```javascript
/**
 * 预加载下一页数据
 */
const preloadNextPage = () => {
  const nextPage = currentPage.value + 1
  // 在用户滚动到80%时预加载
  if (nextPage * pageSize < total.value) {
    loadMoreDishes(nextPage, true)
  }
}
```

- 首屏数据加载时间减少 60%
- 用户感知的加载速度显著提升

```javascript
/**
 * 数据分析埋点工具
 */
class Analytics {
  constructor() {
    this.baseUrl = '/api/v1/analytics'
    this.queue = []
    this.isUploading = false
  }

  /**
   * 页面访问
   */
  trackPageView(pageName, properties = {}) {
    this.track('page_view', {
      page_name: pageName,
      timestamp: Date.now(),
      ...properties
    })
  }

  /**
   * 点击事件
   */
  trackClick(eventName, element, properties = {}) {
    this.track('click', {
      event_name: eventName,
      element_type: element,
      timestamp: Date.now(),
      ...properties
    })
  }

  /**
   * 曝光事件
   */
  trackImpression(eventName, properties = {}) {
    this.track('impression', {
      event_name: eventName,
      timestamp: Date.now(),
      ...properties
    })
  }

  /**
   * 业务事件
   */
  trackEvent(eventName, properties = {}) {
    this.track(eventName, {
      timestamp: Date.now(),
      ...properties
    })
  }

  /**
   * 发送埋点数据
   */
  async track(eventType, properties) {
    // 添加通用属性
    const data = {
      event_type: eventType,
      user_id: this.getUserId(),
      session_id: this.getSessionId(),
      platform: this.getPlatform(),
      device_id: this.getDeviceId(),
      properties
    }

    // 添加到队列
    this.queue.push(data)

    // 批量上传
    if (this.queue.length >= 10) {
      await this.flush()
    }
  }

  /**
   * 批量上传埋点数据
   */
  async flush() {
    if (this.isUploading || this.queue.length === 0) return

    this.isUploading = true

    try {
      const events = this.queue.splice(0, 50)

      await uni.request({
        url: this.baseUrl + '/batch',
        method: 'POST',
        data: { events }
      })

      console.log(`✓ 埋点上传成功: ${events.length}条`)
    } catch (error) {
      console.error('埋点上传失败:', error)
      // 失败的数据放回队列
      this.queue.unshift(...events)
    } finally {
      this.isUploading = false
    }
  }

  /**
   * 获取用户ID
   */
  getUserId() {
    const userStore = require('@/store').useUserStore()
    return userStore.userInfo?.userId || 'anonymous'
  }

  /**
   * 获取会话ID
   */
  getSessionId() {
    let sessionId = uni.getStorageSync('sessionId')
    if (!sessionId) {
      sessionId = this.generateUUID()
      uni.setStorageSync('sessionId', sessionId)
    }
    return sessionId
  }

  /**
   * 获取平台信息
   */
  getPlatform() {
    const systemInfo = uni.getSystemInfoSync()
    return {
      os: systemInfo.platform,
      app: systemInfo.app,
      version: systemInfo.version
    }
  }

  /**
   * 获取设备ID
   */
  getDeviceId() {
    let deviceId = uni.getStorageSync('deviceId')
    if (!deviceId) {
      deviceId = this.generateUUID()
      uni.setStorageSync('deviceId', deviceId)
    }
    return deviceId
  }

  /**
   * 生成UUID
   */
  generateUUID() {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
      const r = Math.random() * 16 | 0
      const v = c === 'x' ? r : (r & 0x3 | 0x8)
      return v.toString(16)
    })
  }
}

// 创建单例
const analytics = new Analytics()

// 页面卸载时上传
uni.onAppHide(() => {
  analytics.flush()
})

export default analytics
```

```javascript
    this.clicks = new Map()
  }

  /**
   * 记录曝光
   */
  recordImpression(dishId, recommendSource) {
    const key = `${dishId}_${recommendSource}`

    if (!this.impressions.has(key)) {
      this.impressions.set(key, {
        dishId,
        recommendSource,
        count: 0,
        firstSeen: Date.now()
      })
    }

    this.impressions.get(key).count++
  }

  /**
   * 记录点击
   */
  recordClick(dishId, recommendSource) {
    const key = `${dishId}_${recommendSource}`

    if (!this.clicks.has(key)) {
      this.clicks.set(key, {
        dishId,
        recommendSource,
        count: 0
      })
    }

    this.clicks.get(key).count++
  }

  /**
   * 计算点击率
   */
  calculateCTR(dishId, recommendSource) {
    const key = `${dishId}_${recommendSource}`
    const impression = this.impressions.get(key)
    const click = this.clicks.get(key)

    if (!impression || impression.count === 0) return 0

    return (click?.count || 0) / impression.count
  }

  /**
   * 上报分析数据
   */
  async report() {
    const data = {
      impressions: Array.from(this.impressions.values()),
      clicks: Array.from(this.clicks.values()),
      timestamp: Date.now()
    }

    try {
      await uni.request({
        url: '/api/v1/recommendation/analytics',
        method: 'POST',
        data
      })

      // 清空已上报的数据
      this.impressions.clear()
      this.clicks.clear()

      console.log('✓ 推荐分析数据已上报')
    } catch (error) {
      console.error('上报推荐分析数据失败:', error)
    }
  }
}

export default new RecommendationAnalytics()
```
