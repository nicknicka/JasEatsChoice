// NOTIF-005: 加载通知详情
const loadNotificationDetail = async () => {
  // NOTIF-005: 调用API获取通知详情
  const res = await notificationApi.getDetail(notificationId.value)

  if (res.code === 200 && res.data) {
    const data = res.data

    // 解析额外信息
    let extra = null
    try {
      extra = typeof data.extra === 'string' ? JSON.parse(data.extra) : data.extra
    } catch (e) {
      extra = data.extra
    }

    notificationDetail.value = {
      id: data.id,
      type: data.type || 'system',
      title: data.title || '通知',
      content: data.content || '',
      isRead: data.isRead || false,
      time: formatTime(data.createdAt),
      extra: extra,
      action: data.action || null,
      relatedList: data.relatedList || []
    }

    // NOTIF-006: 如果未读，标记为已读
    if (!data.isRead) {
      await markAsRead()
    }
  }
}

// NOTIF-006: 标记为已读
const markAsRead = async () => {
  // NOTIF-006: 调用API标记为已读
  const res = await notificationApi.markAsRead(notificationId.value, {
    userId: userId.value

export const notificationApi = {
  // 获取通知列表
  getList: (params) => get('/v1/notifications', params),
  
  // 获取通知详情
  getDetail: (notificationId) => get(`/v1/notifications/${notificationId}`),

  markAsRead: (notificationId, data) => post(`/v1/notifications/${notificationId}/read`, data),
  
  // 批量标记已读
  batchMarkAsRead: (data) => post('/v1/notifications/batch-read', data),
  
  // 全部标记已读
  markAllAsRead: (data) => post('/v1/notifications/mark-all-read', data),
  
  // 删除通知
  delete: (notificationId, params) => del(`/v1/notifications/${notificationId}`, params),
  
  // 批量删除
  batchDelete: (data) => post('/v1/notifications/batch-delete', data),
  
  // 清空通知
  clear: (data) => post('/v1/notifications/clear', data)
}

export const reviewApi = {
  // 发表评价
  create: (data) => post('/v1/reviews', data),
  
  // 获取菜品评价列表
  getDishReviews: (dishId, params) => get(`/v1/reviews/dishes/${dishId}`, params),

  getUserReviews: (userId, params) => get(`/v1/reviews/users/${userId}`, params),
  
  // 获取评价详情
  getDetail: (reviewId) => get(`/v1/reviews/${reviewId}`),
  
  // 商家回复评价
  reply: (reviewId, data) => post(`/v1/reviews/${reviewId}/reply`, data),
  
  // 删除评价
  delete: (reviewId, userId) => del(`/v1/reviews/${reviewId}`, { userId }),
  
  // 上传评价图片
  uploadImages: (formData) => post('/v1/reviews/images', formData),
  
  // 点赞评价

  unlike: (reviewId, data) => post(`/v1/reviews/${reviewId}/unlike`, data),

  getStatistics: (params) => get('/v1/reviews/statistics', params),
  
  // 获取评价标签

}
