## 🚀 使用示例

### API使用

```javascript
// 导入API
import api from '@/api'

// 用户登录
const login = async () => {
  const res = await api.user.loginByPhone({
    phone: '13800138000',
    code: '123456'
  })
  console.log(res.data)
}

// 获取菜品列表
const getDishes = async () => {
  const res = await api.dish.getDishList({
    page: 1,
    pageSize: 10,
    categoryId: 1
  })
  console.log(res.data)
}

// 创建订单
const createOrder = async () => {
  const res = await api.order.createOrder({
    addressId: 1,
    items: [{ dishId: 1, quantity: 2 }]
  })
  console.log(res.data)
}
```
